\pgfdeclarelayer{fondoInterno}

% Parámetros configurables de diseño
\providecommand{\padBandaX}{0.40cm}      % Margen horizontal de las bandas respecto al contenido
\providecommand{\padBandaY}{0.25cm}      % Margen vertical interno de cada banda (arriba y abajo)
\providecommand{\padSubX}{\padBandaX}    % Margen horizontal de bandaSub (por defecto hereda \padBandaX)
\providecommand{\padSubY}{\padBandaY}    % Margen vertical de bandaSub (por defecto hereda \padBandaY)
\providecommand{\sepBandas}{0.55cm}      % Separación vertical exterior entre bandas
\providecommand{\anchoFlanco}{1.80cm}    % Grosor horizontal de las barras laterales
\providecommand{\sepFlanco}{0.50cm}      % Separación horizontal entre bandas y flancos laterales
\providecommand{\padTextoFlanco}{0.55cm} % Margen interno para el ajuste del texto vertical

\tikzset{
    pad x/.store in=\padActualX,
    pad y/.store in=\padActualY,
    pad x=\padBandaX,
    pad y=\padBandaY,
    extra sep fila/.store in=\extraSepFila,
    extra sep fila=0pt,
    envolvente/.style={inner sep=0pt, outer sep=0pt, minimum size=0pt, fit={#1}},
    cajaBase/.style={
        rectangle, rounded corners=3pt, thick, minimum height=1.15cm,
        inner xsep=3.5pt, align=center, font=\scriptsize\bfseries\sffamily
    },
    % #1: color | #2: ancho | #3: opciones extra
    caja/.style n args={3}{cajaBase, draw=#1!80!black, fill=#1!8, text width=#2, #3},
    % #1: color | #2: grosor
    flecha/.style 2 args={->, line width=#2, draw=#1!75!black},
    flechaFlujo/.style={flecha={teal}{3pt}, shorten <=2pt, shorten >=2pt},
    % Estilos de contenedores de fondo y cabeceras
    banda/.style={rectangle, rounded corners=5pt, fill=#1!3, draw=#1!45!black, thick, inner sep=0pt},
    tituloBanda/.style={
        fill=#1!75!black, text=white, font=\bfseries\footnotesize\sffamily,
        rounded corners=2pt, inner xsep=6pt, inner ysep=3pt, outer sep=0pt
    },
    % Estilo de flancos laterales
    flanco/.style={
        cajaBase, rounded corners=4pt, draw=black!70, fill=gray!18,
        line width=1.1pt, minimum width=\anchoFlanco
    },
    % Estilos puntuales por ID (activados automáticamente con /.try)
    mis2/.style={fill=teal!22, draw=teal!95!black},
    bandaSub/.style={densely dashed, line width=1.1pt, pad x=\padSubX, pad y=\padSubY},
    filaSop/.style={extra sep fila=\padSubY}
}

\providecommand{\extraerAncho}[3]{#2}

% #1: args de flecha (opc) | #2: id de fila | #3: args de caja | #4: sep | #5: lista
\providecommand{\filaCajas}[5][]{%
    \foreach \item [count=\n] in {#5} {\xdef\numCajas{\n}}%
    \pgfmathsetmacro{\xIni}{-0.5 * (\numCajas - 1) * (\extraerAncho#3 + 7.8pt + #4)}%
    \begin{scope}[local bounding box=#2, #2/.try]
        \pgfmathsetlengthmacro{\saltoY}{\sepBandas + 2*\padBandaY + \extraSepFila + 1em + 6pt}%
        \foreach \id/\texto [count=\idx, remember=\id as \prev] in {#5} {
            \tikzset{nodoActual/.style={caja=#3, \id/.try}}
            \ifnum\idx=1
                \ifx\filaAnterior\empty
                    \node[nodoActual, anchor=north] (\id) at (\xIni pt, 0) {\texto};
                \else
                    \node[nodoActual, anchor=north] (\id) at ([xshift=\xIni pt, yshift=-\saltoY] 0,0 |- \filaAnterior.south) {\texto};
                \fi
            \else
                \node[nodoActual, right=#4 of \prev] (\id) {\texto};
                \if\relax\detokenize{#1}\relax\else
                    \draw[flecha=#1] (\prev) -- (\id);
                \fi
            \fi
        }%
    \end{scope}%
    \gdef\filaAnterior{#2}%
}

% #1: capa (opc) | #2: id | #3: color | #4: nodos contenido | #5: ref horizontal | #6: título
\providecommand{\dibujarBanda}[6][background]{%
    \begin{scope}[#2/.try]
        \node[envolvente={#4}] (#2-cont) {};
        \node[tituloBanda=#3, anchor=south west] (#2-tit)
            at ([xshift=-\padActualX, yshift=\padActualY] #5.west |- #2-cont.north) {#6};
        \begin{pgfonlayer}{#1}
            \node[banda=#3, #2/.try,
                  fit={(#2-tit)([xshift=\padActualX, yshift=-\padActualY] #5.east |- #2-cont.south)}] (#2) {};
        \end{pgfonlayer}%
    \end{scope}%
}

% #1: lado (left/right) | #2: id | #3: ángulo | #4: título | #5: subtítulo
\providecommand{\dibujarFlanco}[5]{%
    \path let \p1 = ($(bloqueBandas.north) - (bloqueBandas.south)$),
              \n1 = {\y1 - 2*\padTextoFlanco} in
        node[flanco, minimum height=\y1, #1=\sepFlanco of bloqueBandas] (#2) {%
            \rotatebox{#3}{\parbox{\n1}{%
                \centering\bfseries\small #4\\[0.1cm]\footnotesize\normalfont #5%
            }}%
        };
}
