\subsubsection{Subproceso 3.B: Confirmaci\'on y Registro Oficial}

\textbf{Personas:}
\begin{itemize}
    \item \textbf{Estudiante Regular:} Receptor de los documentos oficiales del cierre; verifica la exactitud de sus datos y emplea la Boleta visada en los tr\'amites que la exigen.
    \item \textbf{Personal de la Oficina de Estad\'istica de la Facultad:} Encargado de la depuraci\'on del padr\'on, la remisi\'on del archivo digital de consolidaci\'on, el visado del Reporte y la Boleta, y la formalizaci\'on del cierre ante la DIRCE.
    \item \textbf{Personal de la DIRCE:} Responsable del registro definitivo en el SIGA, de la generaci\'on de boletas oficiales, de la consolidaci\'on del padr\'on nacional y de la transferencia de datos a SUNEDU y MINEDU.
\end{itemize}

\textbf{Tecnolog\'ia:}
\begin{itemize}
    \item \textbf{Sistema Integrado de Gesti\'on Acad\'emica (SIGA):} Plataforma central donde se asienta la matr\'icula con car\'acter legal definitivo y se custodia el historial acad\'emico formal del estudiante.
    \item \textbf{Intranet DIRCE (\texttt{https://dirce.uni.edu.pe/}):} Portal institucional donde se bloquea la fase de edici\'on, se emiten los comprobantes del registro y se consulta el estado de la matr\'icula del periodo.
    \item \textbf{M\'odulo de firma y visado digital:} Mecanismo disponible en la Intranet DIRCE que certifica la validez legal del Reporte y de la Boleta de Matr\'icula una vez aplicado el visado de la Oficina de Estad\'istica.
    \item \textbf{Plataforma de transferencia de datos con SUNEDU:} Canal de transferencia electr\'onica de datos que sustenta la remisi\'on del padr\'on nacional de matriculados y la habilitaci\'on de la carnetizaci\'on estudiantil.
\end{itemize}

\textbf{Documentos y referencias:}
\begin{itemize}
    \item \textbf{Reglamento de Matr\'icula (RR-0570-2022):} Su Cap\'itulo VIII (arts. 75\textdegree{} a 80\textdegree{}) regula la verificaci\'on por secci\'on, las rectificaciones de la primera semana, el Cierre de Matr\'icula, la oportunidad de los padrones, los reportes externos y la devoluci\'on del registro a las Facultades; su Disposici\'on Transitoria \'Unica autoriza la simplificaci\'on y automatizaci\'on de los tr\'amites de matr\'icula.
    \item \textbf{Ficha TUPA del procedimiento de Matr\'icula Regular (PA5400954D, RR-3698-2024):} Documento oficial que establece la condici\'on de aprobaci\'on autom\'atica del tr\'amite, su atenci\'on en dos d\'ias h\'abiles y su car\'acter gratuito, con base legal en la Ley Universitaria N.\textsuperscript{o} 30220 (arts. 41, 42 y 128), el Estatuto de la UNI y las resoluciones RR-0570-2022 y RR-2461-2023.
\end{itemize}
