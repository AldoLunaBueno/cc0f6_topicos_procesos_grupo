\subsubsection{Subproceso 3.B: Confirmación y Registro Oficial}

\textbf{Personas:}
\begin{itemize}
    \item \textbf{Estudiante Regular:} Recibe los documentos finales de su matrícula, revisa que sus datos estén correctos y usa la Boleta firmada en los trámites que la piden.
    \item \textbf{Personal de la Oficina de Estadística de la Facultad:} Revisa y cierra el padrón de la Facultad, envía el archivo digital final a la DIRCE, firma el Reporte y la Boleta de Matrícula, y comunica el fin del proceso.
    \item \textbf{Personal de la DIRCE:} Registra la matrícula definitiva en el SIGA, emite las boletas oficiales, consolida el padrón general de la universidad y envía los datos a la SUNEDU y al MINEDU.
\end{itemize}

\textbf{Tecnología:}
\begin{itemize}
    \item \textbf{Sistema Integrado de Gestión Académica (SIGA):} Base de datos central donde se guarda la matrícula definitiva y el historial académico oficial de cada alumno.
    \item \textbf{Intranet DIRCE (\url{https://dirce.uni.edu.pe/}):} Portal web donde se cierra la edición de cursos, se descargan los comprobantes de matrícula y se consulta el estado final del registro.
    \item \textbf{Módulo de firma y visado digital:} Herramienta dentro de la Intranet DIRCE con la que la Oficina de Estadística firma el Reporte y la Boleta de Matrícula para darles validez oficial.
    \item \textbf{Plataforma de envío de datos a SUNEDU:} Sistema digital mediante el cual la universidad envía el padrón de matriculados para el registro nacional y la emisión del carné universitario.
\end{itemize}

\textbf{Documentos y referencias:}
\begin{itemize}
    \item \textbf{Reglamento de Matrícula (RR-0570-2022):} Su Capítulo VIII (arts. 75\textdegree{} a 80\textdegree{}) regula la revisión de inscritos por sección, las rectificaciones de la primera semana, el Cierre de Matrícula, la entrega de listas a los docentes, los reportes al Estado y la entrega del registro final a las Facultades. Además, su Disposición Transitoria Única autoriza simplificar y automatizar los trámites de matrícula.
    \item \textbf{Ficha TUPA de Matrícula Regular (PA5400954D, RR-3698-2024):} Documento oficial que indica que la matrícula es gratuita, de aprobación automática y se atiende en un máximo de dos días hábiles, según la Ley Universitaria N.\textsuperscript{o} 30220, el Estatuto de la UNI y los reglamentos vigentes.
\end{itemize}
