\subsubsection{Subproceso 3.B: Confirmación y Registro Transaccional Oficial}

\textbf{Personas:}
\begin{itemize}
    \item \textbf{Estudiante Regular:} Receptor de los documentos oficiales del cierre; verifica la exactitud de sus datos y emplea la Boleta visada en los trámites que la exigen.
    \item \textbf{Personal de la Oficina de Estadística de la Facultad:} Encargado de la depuración del padrón, la remisión del archivo digital de consolidación, el visado del Reporte y la Boleta, y la formalización del cierre ante la DIRCE.
    \item \textbf{Personal de la DIRCE:} Responsable del asentamiento transaccional en el SIGA, de la generación de boletas oficiales, de la consolidación del padrón nacional y de la interoperabilidad con SUNEDU y MINEDU.
\end{itemize}

\textbf{Tecnología:}
\begin{itemize}
    \item \textbf{Sistema Integrado de Gestión Académica (SIGA):} Base de datos relacional y plataforma transaccional donde se asienta la matrícula con carácter legal y se custodia el historial académico formal del estudiante.
    \item \textbf{Intranet DIRCE (\texttt{https://dirce.uni.edu.pe/}):} Portal institucional mediante el cual se bloquea la fase de edición, se emiten los comprobantes del registro y se consulta el estado de la matrícula del periodo.
    \item \textbf{Módulo de firma y visado digital:} Mecanismo disponible en la Intranet DIRCE que certifica la validez legal del Reporte y de la Boleta de Matrícula una vez aplicado el visado de la Oficina de Estadística.
    \item \textbf{Plataforma de interoperabilidad con SUNEDU:} Canal de transferencia electrónica de datos que sustenta la remisión del padrón nacional de matriculados y la habilitación de la carnetización estudiantil.
\end{itemize}

\textbf{Documentos y referencias:}
\begin{itemize}
    \item \textbf{Reglamento de Matrícula (RR-0570-2022):} Su Capítulo VIII (arts. 75° a 80°) regula la verificación por sección, las rectificaciones de la primera semana, el Cierre de Matrícula, la oportunidad de los padrones, los reportes externos y la devolución del registro a las Facultades; su Disposición Transitoria Única autoriza la simplificación y automatización de los trámites de matrícula.
    \item \textbf{Ficha TUPA del procedimiento de Matrícula Regular (PA5400954D, RR-3698-2024):} Documento oficial que establece la condición de aprobación automática del trámite, su atención en dos días hábiles y su carácter gratuito, con base legal en la Ley Universitaria N.\textsuperscript{o} 30220 (arts. 41, 42 y 128), el Estatuto de la UNI y las resoluciones RR-0570-2022 y RR-2461-2023.
\end{itemize}
