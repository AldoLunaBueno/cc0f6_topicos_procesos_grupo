\subsubsection{Subproceso 3.A: Verificaci\'on y Control de Reglas Acad\'emicas}

\begin{itemize}
    \item \textbf{Sistema SIGA-ORCE:} Primer nivel de validaci\'on transversal (de pre-fichas).

    \item \textbf{Direcci\'on de Escuela:} Responsable de emitir dict\'amenes de rectificaci\'on (adici\'on, reducci\'on, cambio de grupo).

    \item \textbf{Comisi\'on de Matr\'icula de la Facultad:} Ayuda en la resoluci\'on de casos especiales (cruce de horarios, ampliaci\'on de vacantes si es posible).

    \item \textbf{Estudiante Regular:} tanto los pasivo (espera a que se valide) o activo (si necesita realizar alguna rectificaci\'on).
\end{itemize}
