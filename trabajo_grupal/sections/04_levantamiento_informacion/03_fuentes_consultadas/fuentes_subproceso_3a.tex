\subsubsection{Subproceso 3.A: Verificación y Control de Reglas Académicas}

\textbf{Fuentes documentales:}
\begin{itemize}
    \item \textbf{Reglamento de Matrícula (RR-0570-2022):} Norma de la universidad que define los requisitos de los planes de estudio, los motivos de observación de la Pre-Ficha y las reglas de rectificación.
    \item \textbf{Directivas internas de la Facultad:} Indicaciones de la Facultad sobre el control de estudiantes en riesgo académico (Grupo Cero) y la atención de casos especiales.
\end{itemize}

\textbf{Fuentes humanas:}
\begin{itemize}
    \item \textbf{Personal administrativo de la Dirección de Escuela y Comisión de Matrícula:} Encargados de revisar manualmente los casos observados y de evaluar y resolver las solicitudes de rectificación.
\end{itemize}
