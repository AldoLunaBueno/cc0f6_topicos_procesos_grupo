\subsubsection{Subproceso 3.A: Verificaci\'on y Control de Reglas Acad\'emicas}

\textbf{Fuentes documentales:}
\begin{itemize}
    \item \textbf{Reglamento de Matr\'icula (RR-0570-2022):} Norma institucional que establece las disposiciones generales, los requisitos de los planes de estudio, las causales de observaci\'on y las reglas normativas aplicables al proceso de auditor\'ia.
    \item \textbf{Directivas internas de la facultad:} Documentos normativos y lineamientos espec\'ificos que establecen los criterios y procedimientos institucionales para el manejo y control de los alumnos en riesgo acad\'emico (Grupo Cero).
\end{itemize}

\textbf{Fuentes humanas:}
\begin{itemize}
    \item \textbf{Personal administrativo de la direcci\'on de escuela y comisi\'on de matr\'icula:} Actores y autoridades responsables de la auditor\'ia manual del proceso, as\'i como de la evaluaci\'on, gesti\'on y resoluci\'on de las solicitudes y dict\'amenes de rectificaci\'on.
\end{itemize}
