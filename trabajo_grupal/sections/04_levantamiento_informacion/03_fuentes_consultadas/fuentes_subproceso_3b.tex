\subsubsection{Subproceso 3.B: Confirmaci\'on y Registro Oficial}

\textbf{Fuentes documentales:}
\begin{itemize}
    \item \textbf{Reglamento de Matr\'icula (RR-0570-2022):} Norma institucional cuyo Cap\'itulo VIII (arts. 75\textdegree{} a 80\textdegree{}) regula la verificaci\'on por secci\'on, las rectificaciones de la primera semana, el Cierre de Matr\'icula, la oportunidad de los padrones, el reporte a SUNEDU y MINEDU y la devoluci\'on del registro a las Facultades; incluye la Disposici\'on Transitoria \'Unica que autoriza la automatizaci\'on de los tr\'amites de matr\'icula.
    \item \textbf{Ficha TUPA del procedimiento de Matr\'icula Regular (PA5400954D, RR-3698-2024):} Documento oficial que establece la aprobaci\'on autom\'atica del tr\'amite, la atenci\'on en dos d\'ias h\'abiles y su car\'acter gratuito.
    \item \textbf{Ley Universitaria N.\textsuperscript{o} 30220 y Estatuto de la UNI:} Marco legal e institucional general del registro acad\'emico universitario, base legal del procedimiento seg\'un la ficha TUPA.
\end{itemize}

\textbf{Fuentes tecnol\'ogicas:}
\begin{itemize}
    \item \textbf{Intranet DIRCE (\texttt{https://dirce.uni.edu.pe/}):} Portal institucional observado para registrar el flujo de comprobantes del registro, el estado de la matr\'icula del periodo y los mensajes del sistema durante el cierre.
    \item \textbf{Sistema SIGA:} Plataforma de gesti\'on acad\'emica donde se infiri\'o el proceso de asentamiento y cierre definitivo de la matr\'icula, a partir de la observaci\'on del comportamiento del portal y de la documentaci\'on institucional disponible.
\end{itemize}

\textbf{Fuentes humanas:}
\begin{itemize}
    \item \textbf{Registro de campo del periodo 2026-2 aportado por un integrante del equipo:} Anotaciones elaboradas a partir de la experiencia directa del proceso en el periodo 2026-2 y de comentarios recabados de estudiantes y de la comunidad de Facultad. Esta fuente respalda el diagn\'ostico del cierre, en particular la comunicaci\'on Facultad--DIRCE al concluir la matr\'icula y la falta de acceso de la Facultad para realizar modificaciones en el registro.
\end{itemize}
