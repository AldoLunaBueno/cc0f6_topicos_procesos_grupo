\subsubsection{Subproceso 3.B: Confirmación y Registro Oficial}

\textbf{Fuentes documentales:}
\begin{itemize}
    \item \textbf{Reglamento de Matrícula (RR-0570-2022):} Su Capítulo VIII (arts. 75\textdegree{} a 80\textdegree{}) regula la revisión de inscritos por sección, las rectificaciones de la primera semana, el Cierre de Matrícula, la entrega de listas a los profesores, el reporte a SUNEDU y MINEDU y la entrega del registro final a las Facultades; además, su Disposición Transitoria Única autoriza automatizar los trámites de matrícula.
    \item \textbf{Ficha TUPA de Matrícula Regular (PA5400954D, RR-3698-2024):} Documento oficial que establece la aprobación automática del trámite, su atención en dos días hábiles y su gratuidad.
    \item \textbf{Ley Universitaria N.\textsuperscript{o} 30220 y Estatuto de la UNI:} Base legal general del registro de matrícula en la universidad.
\end{itemize}

\textbf{Fuentes tecnológicas:}
\begin{itemize}
    \item \textbf{Intranet DIRCE (\url{https://dirce.uni.edu.pe/}):} Portal web revisado para ver cómo se emiten los comprobantes de matrícula y cómo cambia el estado del trámite al cerrarse el proceso.
    \item \textbf{Sistema SIGA:} Sistema central de gestión académica donde se guarda el registro definitivo y se realiza el cierre de la matrícula.
\end{itemize}

\textbf{Fuentes humanas:}
\begin{itemize}
    \item \textbf{Apuntes de campo del ciclo 2026-2 tomados por un integrante del equipo:} Notas basadas en la experiencia directa durante la matrícula 2026-2 y en comentarios de alumnos y personal de la Facultad, donde se documenta cómo la Facultad comunica el cierre a la DIRCE y la falta de permisos en el sistema para hacer cambios directos desde la Facultad.
\end{itemize}
